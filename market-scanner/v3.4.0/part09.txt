 state.seenSerial += 1
    record.lastSeen = now
    record.orderAt = state.seenSerial
    state.seenOrder[#state.seenOrder + 1] = {
        key = key,
        at = state.seenSerial,
    }
    pruneSeen(now)
    return record, isNew
end

        urgentPrice,
        record
    )
    local sent, sendErr = sendWebhookPayload(payload)
    if sent then
        if reminder then
            state.stockReminders += 1
            record.lastReminderAt = os.time()
            addLog(
                "DISCORD",
                "재고 유지 확인 일반 Webhook 전송 성공 | "

local function scanCondition(item, star, starLabel)
    state.current = item.ko .. " · " .. starLabel

    detailLog(
        "SCAN",
        tostring(item.ko) .. " | " .. tostring(starLabel) .. " 조회 시작"
    )

    local result, matchedName, matchedMode, err, failureKind =
        findItemResult(item, star, starLabel)

    state.scans += 1

    if err then
        state.lastError = err
        if failureKind == "STOPPED" then
            return
        end
        if failureKind ~= "COOLDOWN" then
            state.errors += 1
        end
        markUnhealthyQuery(failureKind, err)
        return
    end

    if not result then
        markHealthyQuery()
        detailLog(
            "MISS",
            tostring(item.ko) .. " | " .. tostring(starLabel) .. " | 최종 결과 없음"
        )
        return
    end

    markHealthyQuery()
    state.finds += 1

    addLog(
        "FOUND",
        tostring(matchedMode or "?") .. " 조회 성공"
            .. " | " .. comma(result.Price) .. " 다이아"
            .. " | " .. tostring(item.ko)
            .. " | " .. tostring(starLabel)
            .. " | value=" .. tostring(matchedName or "?")
    )

    local key = resultKey(item, starLabel, result)
    local seenNow = os.time()
    local record, isNew = touchSeen(key, seenNow)
    local urgentPrice = getUrgentPriceForItem(item)
    local urgent = result.Price <= urgentPrice

    if not isNew then
        state.duplicates += 1
        addLog(
            "DUPLICATE",
            "이미 알림 전송한 매물 | "
                .. tostring(item.ko)
                .. " | " .. tostring(starLabel)
                .. " | " .. comma(result.Price)
                .. " | JobId " .. tostring(result.JobId)
        )

        if state.stopRequested or lifecycle.Closed then
            return
        end

        if not record.generalSent then
            sendGeneralListingAlert(item, starLabel, result, urgentPrice, record, false)
        end
        if urgent and not record.urgentSent then
            sendUrgentListingAlert(item, starLabel, result, urgentPrice, record)
        end

        local reminderMinutes = math.max(0, tonumber(config.StockReminderMinutes) or 0)
        local reminderSeconds = reminderMinutes * 60
        if record.generalSent and reminderSeconds > 0
            and seenNow - (tonumber(record.lastReminderAt) or record.firstSeen) >= reminderSeconds then
            sendGeneralListingAlert(item, starLabel, result, urgentPrice, record, true)
        end
        return
    end

    if state.stopRequested or lifecycle.Closed then
        return
    end

    sendGeneralListingAlert(item, starLabel, result, urgentPrice, record, false)
    if urgent and not state.stopRequested and not lifecycle.Closed then
        sendUrgentListingAlert(item, starLabel, result, urgentPrice, record)
    end
end

local function addToScanOrder(key)
    for _, existing in ipairs(config.ScanOrder) do
        if existing == key then
            return
        end
    end
    config.ScanOrder[#config.ScanOrder + 1] = key
end

local function removeFromScanOrder(key)
    for index = #config.ScanOrder, 1, -1 do
        if config.ScanOrder[index] == key then
            table.remove(config.ScanOrder, index)
        end
    end
end

local function selectedItems()
    local out = {}
    local added = {}

    for _, key in ipairs(config.ScanOrder) do
        local item = itemByKey[key]
        if item and config.Selected[key] == true and not added[key] then
            out[#out + 1] = item
            added[key] = true
        end
    end

    for _, item in ipairs(catalog) do
        if config.Selected[item.key] == true and not added[item.key] then
            out[#out + 1] = item
            added[item.key] = true
        end
    end
    return out
end

local function selectedCount()
    local n = 0
    for key, selected in pairs(config.Selected) do
        if selected == true and itemByKey[key] then n += 1 end
    end
    return n
end

local function scannerLoop()
    if state.running or lifecycle.Closed then return end

    local chosen = selectedItems()
    local uiSelectedCount = selectedCount()

    if #chosen == 0 then
        state.lastError = "조회할 장비를 하나 이상 선택해주세요."
        addLog("ERROR", state.lastError)
        return
    end

    state.running = true
    state.stopRequested = false
    state.startedAt = os.time()
    state.current = "TerminalService 준비 중"

    addLog("START", "선택 장비 " .. tostring(#chosen) .. "개")

    if uiSelectedCount ~= #chosen then
        addLog(
            "ERROR",
            "UI 선택 수와 실제 스캔 대상 수 불일치 | UI="
                .. tostring(uiSelectedCount)
                .. " | 실제=" .. tostring(#chosen)
        )
    end

    if config.DetailedLogs then
        local maxList = math.min(30, #chosen)
        for i = 1, maxList do
            local item = chosen[i]
            local starText = "성급 없음"
            if item.starred then
                local labels = {}
                for _, star in ipairs(selectedStars(item)) do
                    labels[#labels + 1] = tostring(star) .. "성"
                end
                starText = #labels > 0 and table.concat(labels, "/") or "성급 선택 없음"
            end
            addLog(
                "START",
                tostring(i) .. ". " .. tostring(item.ko)
                    .. " | Name=" .. tostring(item.name)
                    .. " | ID=" .. tostring(item.id)
                    .. " | " .. starText
            )
        end
        if #chosen > maxList then
            addLog(
                "START",
                "... 외 " .. tostring(#chosen - maxList) .. "개 선택됨"
            )
        end
    end

    local conditionCount = countSelectedConditions(chosen)
    if conditionCount == 0 then
        state.running = false
        state.lastError = "선택된 성급 검색 조건이 없습니다."
        addLog("ERROR", state.lastError)
        return
    end
    local minimumCalls, maximumCalls = countSelectedRequestRange(chosen)
    local expectedCalls = tostring(minimumCalls)
    if maximumCalls ~= minimumCalls then
        expectedCalls = expectedCalls .. "~" .. tostring(maximumCalls)
    end

    addLog(
        "INFO",
        "1사이클 검색 조건 " .. tostring(conditionCount)
            .. "회 | FindItem 예상 " .. expectedCalls
            .. "회 | 기본 대기만 약 "
            .. string.format("%.1f", conditionCount * QUERY_INTERVAL / 60)
            .. "분"
    )

    if #chosen >= 50 then
        addLog(
            "WARN",
            "선택 장비가 많아 특정 아이템 조회까지 오래 걸릴 수 있습니다."
        )
    end

    local service = state.service
    local err = nil

    if not service then
        service, err = waitForTerminalService()
    end

    if not service then
        state.running = false
        state.errors += 1
        state.lastError = err
        addLog("ERROR", err)
        sendHealthAlert("TERMINAL_SERVICE", tostring(err))
        return
    end

    state.service = service

    local hookOk, hookMsg = installFindItemObserver(service)
    if hookOk then
        addLog("HOOK", hookMsg)
    else
        addLog("ERROR", hookMsg)
    end

    addLog("CHECK", "선택 장비 " .. tostring(#chosen) .. "개")
    addLog("CHECK", "검색 조건 " .. tostring(conditionCount) .. "개")
    addLog("CHECK", "예상 FindItem " .. expectedCalls .. "회")
    addLog("CHECK", "TerminalService OK")
    addLog("CHECK", "observer " .. (state.findItemHookInstalled and "OK" or "미설치"))
    addLog("CHECK", "일반 Webhook " .. (config.WebhookUrl ~= "" and "설정됨" or "비어 있음"))
    addLog("CHECK", "긴급 Webhook " .. (config.UrgentWebhookUrl ~= "" and "설정됨" or "비어 있음"))
    addLog("CHECK", "자동 저장 " .. (type(writefile) == "function" and "OK" or "미지원"))
    addLog("CHECK", "저사양 모드 " .. (config.LowResourceMode and "ON" or "OFF"))
    addLog("CHECK", "AFK 유지 " .. (config.AfkKeepAlive and "ON" or "OFF"))
    addLog("CHECK", "상세 로그 " .. (config.DetailedLogs and "ON" or "OFF"))

    while not state.stopRequested
        and not lifecycle.Closed
        and sharedEnvironment[GLOBAL_CONTROLLER_KEY] == lifecycle do
        state.cycle += 1
        chosen = selectedItems()

        local cycleStartClock = os.clock()
        local baseScans = state.scans
        local baseFinds = state.finds
        local baseAlerts = state.alerts
        local baseDuplicates = state.duplicates
        local baseReminders = state.stockReminders
        local baseErrors = state.errors
        local baseRemote = state.remoteCalls
        local baseCooldown = state.cooldownResponses

        for _, item in ipairs(chosen) do
            if state.stopRequested then break end

            if item.starred then
                for _, star in ipairs(selectedStars(item)) do
                    if state.stopRequested then break end

                    scanCondition(item, star, tostring(star) .. "성")

                    local waited = 0
                    while waited < QUERY_INTERVAL and not state.stopRequested do
                        task.wait(0.1)
                        waited += 0.1
                    end
                end
            else
                scanCondition(item, nil, "성급 없음")

                local waited = 0
                while waited < QUERY_INTERVAL and not state.stopRequested do
                    task.wait(0.1)
                    waited += 0.1
                end
            end
        end

        if not state.stopRequested then
            local elapsed = os.clock() - cycleStartClock

            addLog(
                "CYCLE",
                "#" .. tostring(state.cycle)
                    .. " 완료 | 조건 " .. tostring(state.scans - baseScans)
                    .. " | 실제호출 " .. tostring(state.remoteCalls - baseRemote)
                    .. " | 발견 " .. tostring(state.finds - baseFinds)
                    .. " | 신규알림 " .. tostring(state.alerts - baseAlerts)
                    .. " | 중복 " .. tostring(state.duplicates - baseDuplicates)
                    .. " | 재고확인 " .. tostring(state.stockReminders - baseReminders)
                    .. " | 쿨다운 " .. tostring(state.cooldownResponses - baseCooldown)
                    .. " | 오류 " .. tostring(state.errors - baseErrors)
                    .. " | " .. string.format("%.1f초", elapsed)
            )

            state.current = "다음 사이클 대기"

            local waited = 0
            while waited < CYCLE_PAUSE and not state.stopRequested do
                task.wait(0.1)
                waited += 0.1
            end
        end
    end

    state.running = false
    state.current = "정지"
    addLog("INFO", "스캐너 정지")
end

local function startScanner(source)
    if state.running or state.startPending or lifecycle.Closed then
        return false, "이미 실행 중이거나 시작 준비 중입니다."
    end

    local chosen = selectedItems()
    if #chosen == 0 then
        addLog("WARN", "조회 장비를 선택해주세요")
        return false, "선택 장비 없음"
    end
    if countSelectedConditions(chosen) == 0 then
        addLog("WARN", "선택한 성급 검색 조건이 없습니다.")
        return false, "검색 조건 없음"
    end
    if type(config.WebhookUrl) ~= "string" or config.WebhookUrl:gsub("%s", "") == "" then
        addLog("WARN", "일반 Webhook URL을 설정해주세요")
        return false, "일반 Webhook 없음"
    end

    state.startPending = true
    addLog("INFO", (source == "auto" and "자동 스캔 시작 준비" or "수동 스캔 시작 준비"))
    task.spawn(function()
        local okLoop, loopErr = pcall(scannerLoop)
        state.startPending = false

        if not okLoop and not lifecycle.Closed then
            state.running = false
            state.errors += 1
            state.lastError = "scannerLoop 예외: " .. tostring(loopErr)
            addLog("ERROR", state.lastError)
            sendHealthAlert("SCANNER_LOOP", state.lastError)
        end
    end)
    return true, "시작 요청 완료"
end

local function stopScanner()
    state.stopRequested = true
    addLog("INFO", "정지 요청")
end
