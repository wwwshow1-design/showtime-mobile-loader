-- 620x450 창모드에서 실제 게임 콘텐츠 영역(약 620x420)을 기준으로 유지.
local uiScale = Instance.new("UIScale")
uiScale.Scale = 1
uiScale.Parent = frame

local camera = workspace.CurrentCamera
local function refreshScale()
    if not camera then
        uiScale.Scale = 1
        return
    end
    local v = camera.ViewportSize
    local sx = v.X / 620
    local sy = v.Y / 420
    uiScale.Scale = math.min(1, sx, sy)
end
refreshScale()
if camera then
    trackConnection(camera:GetPropertyChangedSignal("ViewportSize"):Connect(refreshScale))
end

-- Window drag
do
    local dragging = false
    local dragStart
    local startPos
    local dragInput

    local function update(input)
        if not dragging then return end
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end

    title.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position

            local changed
            changed = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if changed then changed:Disconnect() end
                end
            end)
        end
    end)

    title.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    trackConnection(UserInputService.InputChanged:Connect(function(input)
        if input == dragInput then
            update(input)
        end
    end))
end

trackConnection(UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.End then
        cleanupScanner("End")
    end
end))

local itemButtons = {}
local ITEMS_PER_PAGE_V32 = 9

local function clearItemButtons()
    for _, b in ipairs(itemButtons) do
        pcall(function() b:Destroy() end)
    end
    itemButtons = {}
end

local function refreshFiltered()
    filteredItems = {}
    local q = normalizeSearchText(currentSearch)

    for _, item in ipairs(catalog) do
        if currentClass == "All" or item.class == currentClass then
            local hay = normalizeSearchText(
                tostring(item.ko) .. " "
                .. tostring(item.name) .. " "
                .. tostring(item.id)
            )

            if q == "" or string.find(hay, q, 1, true) then
                filteredItems[#filteredItems + 1] = item
            end
        end
    end

    table.sort(filteredItems, function(a, b)
        return string.lower(a.ko) < string.lower(b.ko)
    end)

    local maxPage = math.max(1, math.ceil(#filteredItems / ITEMS_PER_PAGE_V32))
    currentPage = math.clamp(currentPage, 1, maxPage)
end

local function makeItemCard(item, order)
    local selected = config.Selected[item.key] == true

    local b = Instance.new("TextButton")
    b.LayoutOrder = order
    b.BackgroundColor3 = selected and C.CardSelected or C.Card
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.Text = ""
    b.Parent = cardArea
    corner(b, 6)

    local stroke = Instance.new("UIStroke")
    stroke.Color = C.Accent
    stroke.Thickness = 1
    stroke.Transparency = selected and 0.2 or 1
    stroke.Parent = b

    local accent = Instance.new("Frame")
    accent.Position = UDim2.fromOffset(0, 5)
    accent.Size = UDim2.fromOffset(3, 27)
    accent.BackgroundColor3 = C.Accent
    accent.BorderSizePixel = 0
    accent.Visible = selected
    accent.Parent = b
    corner(accent, 2)

    local check = textLabel(
        b,
        selected and "✓" or "",
        UDim2.new(1, -21, 0, 4),
        UDim2.fromOffset(15, 14),
        12,
        true,
        Color3.fromRGB(194, 186, 255)
    )
    check.TextXAlignment = Enum.TextXAlignment.Center

    local ko = textLabel(
        b,
        tostring(item.ko),
        UDim2.fromOffset(8, 3),
        UDim2.new(1, -30, 0, 15),
        12,
        true,
        C.Text
    )
    ko.TextTruncate = Enum.TextTruncate.AtEnd

    local en = textLabel(
        b,
        tostring(item.name),
        UDim2.fromOffset(8, 18),
        UDim2.new(1, -16, 0, 13),
        10,
        false,
        C.Muted
    )
    en.TextTruncate = Enum.TextTruncate.AtEnd

    b.MouseEnter:Connect(function()
        b.BackgroundColor3 = selected
            and Color3.fromRGB(46, 41, 79)
            or C.CardHover
    end)

    b.MouseLeave:Connect(function()
        b.BackgroundColor3 = selected and C.CardSelected or C.Card
    end)

    b.MouseButton1Click:Connect(function()
        local newSelected = not selected
        config.Selected[item.key] = newSelected and true or nil
        if newSelected then
            addToScanOrder(item.key)
        else
            removeFromScanOrder(item.key)
        end
        scheduleConfigSave("장비 선택 변경")
        renderItems()
    end)

    itemButtons[#itemButtons + 1] = b
end

renderItems = function()
    clearItemButtons()
    refreshFiltered()
    paintCategorySelection()

    local maxPage = math.max(1, math.ceil(#filteredItems / ITEMS_PER_PAGE_V32))
    pageInfo.Text =
        "총 " .. tostring(#filteredItems)
        .. "개   |   " .. tostring(currentPage)
        .. " / " .. tostring(maxPage) .. " 페이지"

    local startIndex = (currentPage - 1) * ITEMS_PER_PAGE_V32 + 1
    for i = 0, ITEMS_PER_PAGE_V32 - 1 do
        local item = filteredItems[startIndex + i]
        if item then
            makeItemCard(item, i + 1)
        end
    end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    currentSearch = searchBox.Text or ""
    currentPage = 1
    renderItems()
end)
