local lastLogSignature = ""

local function richEscape(s)
    s = tostring(s or "")
    s = s:gsub("&", "&amp;")
    s = s:gsub("<", "&lt;")
    s = s:gsub(">", "&gt;")
    return s
end

local function renderLogs()
    local logs = state.logs or {}
    local first = logs[1]
    local sig = tostring(#logs)

    if first then
        sig = sig .. "|" .. tostring(first.at)
            .. "|" .. tostring(first.kind)
            .. "|" .. tostring(first.message)
    end

    if sig == lastLogSignature then
        return
    end
    lastLogSignature = sig

    if #logs == 0 then
        logText.Text = '<font color="#6C7481">로그 대기 중...</font>'
        logScroll.CanvasSize = UDim2.fromOffset(0, 18)
        return
    end

    local lines = {}
    local maxRows = math.min(100, #logs)

    for i = 1, maxRows do
        local r = logs[i]
        local kind = tostring(r.kind or "INFO")
        local color = LOG_HEX[kind] or "#8B95A4"

        lines[#lines + 1] =
            '<font color="#6C7481">' .. richEscape(r.at) .. '</font>'
            .. '  <font color="' .. color .. '"><b>[' .. richEscape(kind) .. ']</b></font>'
            .. '  <font color="#EBEEF4">' .. richEscape(r.message) .. '</font>'
    end

    logText.Text = table.concat(lines, "\n")
    local height = math.max(18, maxRows * 15 + 4)
    logText.Size = UDim2.new(1, -10, 0, height)
    logScroll.CanvasSize = UDim2.fromOffset(0, height + 4)
end

-- Overlay helper
local function makeOverlay(name, w, h)
    local old = frame:FindFirstChild(name)
    if old then old:Destroy() end

    local dim = Instance.new("Frame")
    dim.Name = name
    dim.Position = UDim2.fromScale(0, 0)
    dim.Size = UDim2.fromScale(1, 1)
    dim.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    dim.BackgroundTransparency = 0.32
    dim.BorderSizePixel = 0
    dim.ZIndex = 80
    dim.Parent = frame

    local p = Instance.new("Frame")
    p.AnchorPoint = Vector2.new(0.5, 0.5)
    p.Position = UDim2.fromScale(0.5, 0.5)
    p.Size = UDim2.fromOffset(w, h)
    p.BackgroundColor3 = C.Panel
    p.BorderSizePixel = 0
    p.ZIndex = 81
    p.Parent = dim
    corner(p, 9)

    return dim, p
end
