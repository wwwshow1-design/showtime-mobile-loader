    local function moveSelectedItem(item, delta)
        local ordered = selectedItems()
        local index
        for i, selectedItem in ipairs(ordered) do
            if selectedItem.key == item.key then
                index = i
                break
            end
        end
        local target = index and (index + delta) or nil
        if not index or not target or target < 1 or target > #ordered then
            return
        end
        ordered[index], ordered[target] = ordered[target], ordered[index]
        config.ScanOrder = {}
        for _, selectedItem in ipairs(ordered) do
            config.ScanOrder[#config.ScanOrder + 1] = selectedItem.key
        end
        scheduleConfigSave("선택 장비 스캔 순서 변경")
    end

-- 게임 거래소 UI가 실제로 TerminalService.FindItem을 어떤 인수로 호출하는지
-- 관찰하기 위한 1회 설치 작업입니다. FindItem 자체를 추가 호출하지 않습니다.
task.spawn(function()
    local service, err = waitForTerminalService()
    if lifecycle.Closed or sharedEnvironment[GLOBAL_CONTROLLER_KEY] ~= lifecycle then
        return
    end

    if service then
        state.service = state.service or service
        local okHook, hookMsg = installFindItemObserver(service)
        addLog(okHook and "HOOK" or "ERROR", hookMsg)

        if config.AutoStartScanner ~= false then
            if selectedCount() == 0 then
                addLog("AUTO", "자동 시작 대기 | 선택 장비가 없습니다.")
            elseif type(config.WebhookUrl) ~= "string"
                or config.WebhookUrl:gsub("%s", "") == "" then
                addLog("AUTO", "자동 시작 대기 | 일반 Webhook이 비어 있습니다.")
            else
                startScanner("auto")
            end
        else
            addLog("AUTO", "자동 스캔 시작 OFF | 수동 START 대기")
        end
    else
        addLog("ERROR", "FindItem 관찰 준비 실패 | " .. tostring(err))
        sendHealthAlert("TERMINAL_SERVICE", "FindItem 관찰 준비 실패 | " .. tostring(err))
    end
end)
