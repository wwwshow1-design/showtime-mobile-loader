local function comma(n)
    n = tonumber(n)
    if not n then return "?" end
    local s = tostring(math.floor(n))
    while true do
        local new, count = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
        s = new
        if count == 0 then break end
    end
    return s
end

local function safeFind(root, ...)
    local cur = root
    for i = 1, select("#", ...) do
        if not cur then return nil end
        local childName = select(i, ...)
        if type(childName) ~= "string" then return nil end
        cur = cur:FindFirstChild(childName)
    end
    return cur
end

local function saveConfig()
    if type(writefile) ~= "function" then
        return false, "파일 저장 기능(writefile)을 찾지 못했습니다."
    end
    local ok, err = pcall(function()

end

local function loadConfig()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then
        return false
    end
    local okExists, exists = pcall(isfile, CONFIG_FILE)
    if not okExists or not exists then
        return false
    end
    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(CONFIG_FILE))
    end)
    if not ok or type(decoded) ~= "table" then
        return false
    end

    if type(decoded.WebhookUrl) == "string" then
        config.WebhookUrl = decoded.WebhookUrl
    end
    if type(decoded.UrgentWebhookUrl) == "string" then
        config.UrgentWebhookUrl = decoded.UrgentWebhookUrl
    else
        configNeedsMigration = true
    end
    if tonumber(decoded.UrgentPrice) then
        config.UrgentPrice = math.max(0, math.floor(tonumber(decoded.UrgentPrice)))
    end
    if type(decoded.ItemUrgentPrices) == "table" then
        config.ItemUrgentPrices = decoded.ItemUrgentPrices
    else
        configNeedsMigration = true
    end
    if type(decoded.Selected) == "table" then
        config.Selected = decoded.Selected
    end
    if type(decoded.StarFilters) == "table" then
        config.StarFilters = decoded.StarFilters
    else
        configNeedsMigration = true
    end
    if type(decoded.ScanOrder) == "table" then
        config.ScanOrder = decoded.ScanOrder
    else
        configNeedsMigration = true
    end
    if tonumber(decoded.StockReminderMinutes) then
        config.StockReminderMinutes = math.max(0, math.floor(tonumber(decoded.StockReminderMinutes)))
    else
        configNeedsMigration = true
    end
    if type(decoded.DetailedLogs) == "boolean" then
        config.DetailedLogs = decoded.DetailedLogs
    end
    if type(decoded.LowResourceMode) == "boolean" then
        config.LowResourceMode = decoded.LowResourceMode
    else
        config.LowResourceMode = true
        configNeedsMigration = true
    end
    if type(decoded.AfkKeepAlive) == "boolean" then
        config.AfkKeepAlive = decoded.AfkKeepAlive
    else
        config.AfkKeepAlive = true
        configNeedsMigration = true
    end
    if type(decoded.AutoStartScanner) == "boolean" then
        config.AutoStartScanner = decoded.AutoStartScanner
    else
        config.AutoStartScanner = true
        configNeedsMigration = true
    end
    return true
end

local function getRequestFunction()
    if type(request) == "function" then return request end
    if type(http_request) == "function" then return http_request end
    if type(syn) == "table" and type(syn.request) == "function" then return syn.request end
    return nil
end

local function sendWebhookPayloadTo(url, payload, emptyMessage)
    if type(url) ~= "string" or url:gsub("%s", "") == "" then
        return false, emptyMessage or "Discord Webhook URL이 비어 있습니다."
    end
    local req = getRequestFunction()
    if not req then
        return false, "HTTP request 기능을 찾지 못했습니다."
    end

    local ok, response = pcall(function()
        return req({
            Url = url,
            Method = "POST",
            Headers = {["Content-Type"]="application/json"},
            Body = HttpService:JSONEncode(payload),
        })
    end)
    if not ok then
        return false, tostring(response)
    end

    local code = tonumber(response and (response.StatusCode or response.Status or response.status_code))
    if code and code >= 200 and code < 300 then
        return true, "전송 성공"
    end
    if code == nil and response ~= nil then
        return true, "전송 요청 완료"
    end
    return false, "Discord 응답 코드: " .. tostring(code)
end

local function waitForTerminalService()
    if state.service then
        return state.service, nil
    end

    if state.serviceLoading then
        local sharedDeadline = os.clock() + SERVICE_TIMEOUT
        while state.serviceLoading and os.clock() < sharedDeadline and not state.stopRequested do
            task.wait(0.1)
        end
        if state.service then
            return state.service, nil
        end
        return nil, state.lastServiceError ~= ""
            and state.lastServiceError
            or "TerminalService 공동 대기 시간 초과"
    end

    state.serviceLoading = true
    local knitModule = safeFind(
        ReplicatedStorage, "Packages", "_Index",
        "sleitnick_knit@1.5.1", "knit"
    )
    if not knitModule then
        state.serviceLoading = false
        state.lastServiceError = "Knit 모듈 경로를 찾지 못했습니다."
        return nil, "Knit 모듈 경로를 찾지 못했습니다."
    end

    local okRequire, Knit = pcall(require, knitModule)
    if not okRequire or type(Knit) ~= "table" then
        state.serviceLoading = false
        state.lastServiceError = "Knit 모듈을 불러오지 못했습니다."
        return nil, "Knit 모듈을 불러오지 못했습니다."
    end

    local deadline = os.clock() + SERVICE_TIMEOUT
    local lastErr = ""
    while os.clock() < deadline and not state.stopRequested and not lifecycle.Closed do
        local ok, serviceOrErr = pcall(function()
            return Knit.GetService("TerminalService")
        end)
        if ok and serviceOrErr then
            state.service = serviceOrErr
            state.serviceLoading = false
            state.lastServiceError = ""
            return serviceOrErr, nil
        end
        lastErr = tostring(serviceOrErr)
        task.wait(0.5)
    end
    state.serviceLoading = false

    if state.stopRequested or lifecycle.Closed then
        state.lastServiceError = "정지 요청"
        return nil, state.lastServiceError
    end

    state.lastServiceError = "TerminalService 준비 시간 초과: " .. lastErr
    return nil, state.lastServiceError
end


local LOG_LIMIT = 200

local function addLog(kind, message)
    state.logs = state.logs or {}
    table.insert(state.logs, 1, {
        at = os.date("%H:%M:%S"),
        kind = tostring(kind or "INFO"),
        message = tostring(message or ""),
    })

    while #state.logs > LOG_LIMIT do
        table.remove(state.logs)
    end
end

local function detailLog(kind, message)
    if config.DetailedLogs then
        addLog(kind, message)
    end
end

local configSaveState = {
    revision = 0,
    pending = false,
}

local function performConfigSave(context)
    local ok, message = saveConfig()
    if not ok then
        addLog(
            "ERROR",
            "설정 자동 저장 실패"
                .. (context and (" | " .. tostring(context)) or "")
                .. " | " .. tostring(message)
        )
    end
    return ok
end

local function scheduleConfigSave(context)
    configSaveState.revision += 1
    configSaveState.pending = true
    local requestedRevision = configSaveState.revision

    task.delay(CONFIG_SAVE_DEBOUNCE, function()
        if lifecycle.Closed
            or not configSaveState.pending
            or requestedRevision ~= configSaveState.revision then
            return
        end

        configSaveState.pending = false
        performConfigSave(context or "debounce")
    end)
end

local function flushConfigSave(context)
    if not configSaveState.pending then
        return true
    end

    configSaveState.revision += 1
    configSaveState.pending = false
    return performConfigSave(context or "cleanup")
end

local function getPerformanceBaseline()
    local existing = sharedEnvironment[PERFORMANCE_BASELINE_KEY]
    if type(existing) == "table" then
        return existing
    end

    local baseline = {
        fpsCaptured = false,
        fpsCap = nil,
        qualityCaptured = false,
        qualityLevel = nil,

    if type(getfpscap) == "function" then

            baseline.fpsCaptured = true

        return settings().Rendering.QualityLevel
    end)
    if okQuality and qualityLevel ~= nil then
        baseline.qualityCaptured = true
        baseline.qualityLevel = qualityLevel
    end

    sharedEnvironment[PERFORMANCE_BASELINE_KEY] = baseline
    return baseline
end
